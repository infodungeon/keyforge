FROM httpd:alpine

# Expose non-privileged ports (HTTP proxy only, HTTPS handled by Cloudflare or sidecar)
EXPOSE 8080

# Configure Apache to run as non-root
RUN sed -i 's/Listen 80/Listen 8080/g' /usr/local/apache2/conf/httpd.conf && \
    chown -R www-data:www-data /usr/local/apache2/logs /usr/local/apache2/modules

USER www-data

# Ensure SSL, Rewrite, AND PROXY modules are loaded
RUN sed -i \
    -e '/LoadModule ssl_module/s/^#//g' \
    -e '/LoadModule socache_shmcb_module/s/^#//g' \
    -e '/LoadModule rewrite_module/s/^#//g' \
    -e '/LoadModule proxy_module/s/^#//g' \
    -e '/LoadModule proxy_http_module/s/^#//g' \
    -e '/LoadModule proxy_wstunnel_module/s/^#//g' \
    /usr/local/apache2/conf/httpd.conf

# Disable the dummy certificates that ship with the image
RUN sed -i \
    -e '/SSLCertificateFile.*server\.crt/s/^/#/' \
    -e '/SSLCertificateKeyFile.*server\.key/s/^/#/' \
    /usr/local/apache2/conf/extra/httpd-ssl.conf

# Copy your custom Apache configuration
# Updated path: ops/hosts/sites -> hosts/sites
COPY ./hosts/sites/infodungeon.ssl.conf /usr/local/apache2/conf/extra/

# Include custom configuration
RUN echo 'Include conf/extra/infodungeon.ssl.conf' >> /usr/local/apache2/conf/httpd.conf

# Volume mounts (for dynamic content and certificates)
VOLUME /usr/local/apache2/htdocs/
VOLUME /etc/apache2/ssl/

# Start Apache in foreground
CMD ["httpd-foreground"]
